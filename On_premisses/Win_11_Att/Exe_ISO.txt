============================================
Building the image and runing the upgrade
============================================

# mount the disk image
.\psexec.exe \\HOSTNAME -s powershell.exe -Command "Mount-DiskImage -ImagePath 'C:\Path\BrazilianPortuguese_x64.iso'"


# check
.\psexec.exe \\HOSTNAME cmd
D:
dir
find = setup.exe


============================================
Running the setup of the image
============================================

#run
.\PsExec.exe \\HOSTNAME -s powershell -Command "Start-Process 'D:\setup.exe' -ArgumentList '/auto upgrade /quiet /noreboot /eula accept /dynamicupdate disable' -Wait

-----------
VIA CMD
-----------

.\PsExec.exe \\HOSTNAME -s cmd.exe /c "start /wait D:\setup.exe /auto upgrade /quiet"

.\PsExec.exe \\HOSTNAME -i -s cmd.exe /c "start /wait D:\setup.exe /auto upgrade /quiet"
